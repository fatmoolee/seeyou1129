    async function toggleCheckIn(rowId) {

      const person = attendees.find(p => p.rowId === rowId);

      if (!person) return;


      const targetState = !person.checkedIn;

      const nowStr = new Date().toLocaleString('zh-TW', { hour12: false });


      // 樂觀更新 UI，讓同工操作零延遲

      const prevCheckedIn = person.checkedIn;

      const prevTime = person.checkInTime;

      person.checkedIn = targetState;

      person.checkInTime = targetState ? nowStr : '';

      renderAll();


      if (targetState) {

        showToast(`${person.name} 報到成功！`, '✅');

        // 若在後五碼模式下完成報到，0.8秒後自動清空數字方便下一位來賓

        if (currentMode === 'digits' && currentDigits.length === 5) {

          setTimeout(() => {

            clearDigits();

          }, 800);

        }

      } else {

        showToast(`已取消 ${person.name} 的報到`, '↩️');

      }


      if (!apiUrl) {

        saveDemoLocal();

        return;

      }


      // 同步寫入 Google 試算表

      try {

        const res = await fetch(apiUrl, {

          method: 'POST',

          body: JSON.stringify({

            action: 'toggleCheckIn',

            rowId: rowId,

            checkedIn: targetState

          })

        });

        const json = await res.json();

        if (json.status === 'success') {

          person.checkInTime = json.checkInTime;

          renderAll();

        } else {

          throw new Error(json.message);

        }

      } catch (err) {

        // 若寫入失敗則回滾狀態

        person.checkedIn = prevCheckedIn;

        person.checkInTime = prevTime;

        renderAll();

        showToast('同步試算表失敗，請重試', '❌');

      }

    }

    function appendDigit(d) {

      if (currentDigits.length >= 5) return;

      currentDigits += d;

      updateDigitBoxes();

      renderList();

    }


    function backspaceDigit() {

      if (currentDigits.length === 0) return;

      currentDigits = currentDigits.slice(0, -1);

      updateDigitBoxes();

      renderList();

    }


    function clearDigits() {

      currentDigits = '';

      document.getElementById('hiddenDigitInput').value = '';

      updateDigitBoxes();

      renderList();

    }


    function handlePhysicalInput(val) {

      const cleaned = val.replace(/\D/g, '').slice(0, 5);

      currentDigits = cleaned;

      updateDigitBoxes();

      renderList();

    }


    // 支援電腦實體鍵盤直接打字

    document.addEventListener('keydown', (e) => {

      if (currentMode !== 'digits') return;

      // 若正在彈窗輸入文字則不攔截

      if (!document.getElementById('editPhoneModal').classList.contains('hidden') ||

          !document.getElementById('walkInModal').classList.contains('hidden') ||

          !document.getElementById('settingsModal').classList.contains('hidden')) {

        return;

      }


      if (e.key >= '0' && e.key <= '9') {

        appendDigit(e.key);

      } else if (e.key === 'Backspace') {

        backspaceDigit();

      } else if (e.key === 'Escape') {

        clearDigits();

      }

    });


    function updateDigitBoxes() {

      for (let i = 0; i < 5; i++) {

        const box = document.getElementById(`digitBox${i}`);

        const char = currentDigits[i] || '';

        box.textContent = char;

        if (char) {

          box.classList.remove('border-slate-300', 'bg-slate-50');

          box.classList.add('border-indigo-500', 'bg-indigo-50/50', 'text-indigo-700');

        } else {

          box.classList.add('border-slate-300', 'bg-slate-50');

          box.classList.remove('border-indigo-500', 'bg-indigo-50/50', 'text-indigo-700');

        }

      }

    }

    function setFilterStatus(status) {

      filterStatus = status;

      ['all', 'pending', 'checked'].forEach(st => {

        const btn = document.getElementById('filter' + st.charAt(0).toUpperCase() + st.slice(1));

        if (st === status) {

          btn.className = 'px-2.5 py-1 rounded-lg bg-slate-800 text-white font-medium';

        } else {

          btn.className = 'px-2.5 py-1 rounded-lg bg-slate-200 text-slate-700 font-medium';

    }

    function updateStats() {

      const total = attendees.length;

      const checked = attendees.filter(a => a.checkedIn).length;

      const pending = total - checked;

    function renderList() {

      const container = document.getElementById('attendeeList');

      const titleEl = document.getElementById('listTitle');

      let filtered = [...attendees];


      // 1. 狀態篩選

      if (filterStatus === 'pending') filtered = filtered.filter(a => !a.checkedIn);

      if (filterStatus === 'checked') filtered = filtered.filter(a => a.checkedIn);


      // 2. 搜尋條件篩選

      if (currentMode === 'digits') {

        if (currentDigits.length > 0) {

          titleEl.textContent = `後五碼包含「${currentDigits}」的搜尋結果`;

          filtered = filtered.filter(a => {

            const cleanPhone = a.phone.replace(/\D/g, '');

            const last5 = cleanPhone.slice(-5);

            // 支援輸入過程即時比對（例如輸入3碼時比對後五碼開頭或包含）

            return last5.includes(currentDigits);

          });

        } else {

          titleEl.textContent = `全部名單 (${filtered.length} 人)`;

        }

      } else {

        const kw = document.getElementById('textSearchInput').value.trim().toLowerCase();

        if (kw) {

          titleEl.textContent = `關鍵字「${kw}」搜尋結果`;

          filtered = filtered.filter(a =>

            a.name.toLowerCase().includes(kw) ||

            a.phone.includes(kw) ||

            (a.note && a.note.toLowerCase().includes(kw))

          );

        } else {

          titleEl.textContent = `全部名單 (${filtered.length} 人)`;

        }

      }


      if (filtered.length === 0) {

        container.innerHTML = `

          <div class="bg-white rounded-3xl p-8 text-center border border-slate-200">

            <div class="text-3xl mb-2">🔍</div>

            <div class="font-bold text-slate-700">找不到符合的報名資料</div>

            <p class="text-xs text-slate-500 mt-1">可能是來賓當初填錯手機號碼，請切換至「姓名搜尋 / 號碼異常」查找或現場補登。</p>

            <button onclick="switchMode('rescue')" class="mt-4 px-4 py-2 rounded-xl bg-indigo-50 text-indigo-700 font-bold text-xs hover:bg-indigo-100">

              切換至姓名搜尋 / 補登

            </button>

          </div>

        `;

        return;

      }


      container.innerHTML = filtered.map(person => {

        const cleanPhone = person.phone.replace(/\D/g, '');

        const prefix = cleanPhone.length > 5 ? cleanPhone.slice(0, -5) : '';

        const last5 = cleanPhone.slice(-5);


        return `

          <div class="rounded-2xl p-4 border transition flex items-center justify-between gap-3 ${

            person.checkedIn

              ? 'bg-emerald-50/90 border-emerald-300 shadow-xs'

              : 'bg-white border-slate-200 shadow-xs'

          }">

            <div class="min-w-0 flex-1">

              <div class="flex items-center gap-2 flex-wrap">

                <span class="font-black text-lg text-slate-900 truncate">${escapeHtml(person.name)}</span>

                ${person.note ? `<span class="px-2 py-0.5 rounded-md text-xs font-medium bg-slate-200/80 text-slate-700">${escapeHtml(person.note)}</span>` : ''}

                ${person.checkedIn ? `<span class="px-2 py-0.5 rounded-md text-xs font-bold bg-emerald-200 text-emerald-900">已報到</span>` : ''}

              </div>

              

              <div class="mt-1 flex items-center gap-2 text-sm">

                <span class="font-mono text-slate-500">

                  ${escapeHtml(prefix)}<strong class="text-indigo-700 font-black text-base bg-indigo-50 px-1.5 py-0.5 rounded">${escapeHtml(last5 || person.phone)}</strong>

                </span>

                <button onclick="openEditPhoneModal(${person.rowId})" class="text-xs text-slate-400 hover:text-indigo-600 underline">

                  改電話

                </button>

              </div>


              ${person.checkedIn && person.checkInTime ? `

                <div class="text-[11px] text-emerald-700 mt-1 font-medium">

                  🕒 報到時間：${escapeHtml(person.checkInTime)}

                </div>

              ` : ''}

            </div>


            <div class="shrink-0">

              ${person.checkedIn ? `

                <button onclick="toggleCheckIn(${person.rowId})"

                  class="px-4 py-3 rounded-2xl bg-white hover:bg-rose-50 text-rose-600 border border-rose-200 font-bold text-xs sm:text-sm transition active:scale-95">

                  取消報到

                </button>

              ` : `

                <button onclick="toggleCheckIn(${person.rowId})"

                  class="px-5 py-3.5 rounded-2xl bg-emerald-600 hover:bg-emerald-500 text-white font-black text-sm sm:text-base shadow-md shadow-emerald-600/20 transition active:scale-95">

                  確認報到

                </button>

              `}

            </div>

          </div>

        `;

      }).join('');

    }

    async function submitWalkIn() {

      const name = document.getElementById('walkInName').value.trim();

      const phone = document.getElementById('walkInPhone').value.trim();

      const note = document.getElementById('walkInNote').value.trim() || '現場補登';


      if (!name || !phone) {

        showToast('請填寫姓名與手機號碼', '⚠️');

        return;

      }


      closeModal('walkInModal');

      const nowStr = new Date().toLocaleString('zh-TW', { hour12: false });


      if (!apiUrl) {

        const nextRowId = attendees.length > 0 ? Math.max(...attendees.map(a => a.rowId)) + 1 : 2;

        attendees.unshift({

          rowId: nextRowId,

          name,

          phone,

          note,

          checkedIn: true,

          checkInTime: nowStr

        });

        saveDemoLocal();

        renderAll();

        showToast(`${name} 已現場補登並報到！`, '✅');

        return;

      }


      showToast('正在寫入試算表...', '⏳');

      try {

        const res = await fetch(apiUrl, {

          method: 'POST',

          body: JSON.stringify({

            action: 'addWalkIn',

            name,

            phone,

            note

          })

        });

        const json = await res.json();

        if (json.status === 'success') {

          attendees.unshift(json.record);

          renderAll();

          showToast(`${name} 已同步寫入並完成報到！`, '✅');

        }

      } catch (err) {

        showToast('現場補登寫入失敗', '❌');

      }

    }
